let activeDragWin = null; let dragX = 0, dragY = 0; function toggleStartMenu() { const menu = document.getElementById('startMenu'); menu.style.display = (menu.style.display === 'block') ? 'none' : 'block'; } function openWindow(id) { const win = document.getElementById('win-' + id); win.style.display = 'flex'; focusWindow(win); document.getElementById('startMenu').style.display = 'none'; if(id === 'graphics') { launchPerfRender(); } } function closeWindow(id) { document.getElementById('win-' + id).style.display = 'none'; } function focusWindow(win) { document.querySelectorAll('.window').forEach(w => w.style.zIndex = 10); win.style.zIndex = 100; } function loadWebUrl() { let url = document.getElementById('browser-url').value; if (!url.startsWith('http://') && !url.startsWith('https://')) { url = 'https://' + url; } document.getElementById('browser-frame').src = url; } let animId = null; function launchPerfRender() { const canvas = document.getElementById('perfCanvas'); const ctx = canvas.getContext('webgl') || canvas.getContext('experimental-webgl'); if (!ctx) return; canvas.width = canvas.clientWidth; canvas.height = canvas.clientHeight; function loop(time) { if (document.getElementById('win-graphics').style.display === 'none') { cancelAnimationFrame(animId); return; } let pulse = Math.sin(time * 0.003) * 0.4 + 0.5; ctx.clearColor(0.05 * pulse, 0.1 * pulse, 0.2 * pulse, 1.0); ctx.clear(ctx.COLOR_BUFFER_BIT); animId = requestAnimationFrame(loop); } if(!animId) animId = requestAnimationFrame(loop); } function startDrag(e, id) { activeDragWin = document.getElementById(id); const clientX = e.clientX || (e.touches ? e.touches.clientX : 0); const clientY = e.clientY || (e.touches ? e.touches.clientY : 0); dragX = clientX - activeDragWin.offsetLeft; dragY = clientY - activeDragWin.offsetTop; focusWindow(activeDragWin); document.onmousemove = performDrag; document.ontouchmove = performDrag; document.onmouseup = stopDrag; document.ontouchend = stopDrag; } function performDrag(e) { if (!activeDragWin) return; const clientX = e.clientX || (e.touches ? e.touches.clientX : 0); const clientY = e.clientY || (e.touches ? e.touches.clientY : 0); activeDragWin.style.left = (clientX - dragX) + 'px'; activeDragWin.style.top = (clientY - dragY) + 'px'; } function stopDrag() { activeDragWin = null; document.onmousemove = null; document.ontouchmove = null; } function updateClock() { document.getElementById('system-clock').innerText = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }); } setInterval(updateClock, 1000); updateClock();
